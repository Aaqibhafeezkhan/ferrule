use std::{error::Error, fs};
use codegen_runtime::{FunctionError, RuntimeError, XmlBoundaryError, XmlBoundaryErrorKind};

fn matches(error: &XmlBoundaryError, expected: &str) -> bool {
    if error.kind != XmlBoundaryErrorKind::Mapping { return false; }
    match (expected, error.source().and_then(|source| source.downcast_ref::<RuntimeError>())) {
        ("divide", Some(RuntimeError::Function(FunctionError::DivideByZero))) => true,
        ("cast", Some(RuntimeError::Function(FunctionError::InvalidArgument { function: "coerce_float", message: "requires finite floating-point text" }))) => true,
        _ => false,
    }
}

fn main() -> Result<(), Box<dyn Error>> {
    let cases: Vec<serde_json::Value> = serde_json::from_slice(&fs::read("cases.json")?)?;
    let mut outcomes = Vec::new();
    let mut good = true;
    for case in &cases {
        let input = case["input"].as_str().ok_or("input")?;
        for (api, result) in [
            ("text", ferrule_generated_mapping::execute_xml(input)),
            ("bytes", ferrule_generated_mapping::execute_xml_bytes(input.as_bytes()).map(|bytes| String::from_utf8(bytes).expect("UTF-8 output"))),
        ] {
            good &= match &result {
                Ok(value) => case["error"].is_null() && Some(value.as_str()) == case["expected_xml"].as_str(),
                Err(error) => case["error"].as_str().is_some_and(|expected| matches(error, expected)),
            };
            outcomes.push((case["name"].clone(), api, format!("{result:#?}")));
        }
    }
    fs::write("complete-outcomes.txt", format!("{outcomes:#?}\n"))?;
    assert_eq!(outcomes.len(), cases.len() * 2);
    assert!(good, "complete XML values or exact failure phases differed");
    Ok(())
}
